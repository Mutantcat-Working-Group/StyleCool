# FunctionCool + StyleCool 同服双站点 · 一份总配置
#
#   站点 A：functioncool.mutantcat.org  ->  目录 functioncool/
#   站点 B：stylecool.mutantcat.org     ->  目录 stylecool/
#
# 两个项目分家后各占一个以项目名命名的文件夹，其余规则（伪静态、301、
# PHP 走 FPM）只用这一份总配置，不需要再按站点拆。
#
# 使用方法：
#   1. 把下面的 root 父目录改成面板里「两个站点目录的上一级」；
#      例如面板里分别是 /www/wwwroot/functioncool 和
#      /www/wwwroot/stylecool，父目录就写 /www/wwwroot；
#   2. PHP-FPM socket 换成面板生成的实际地址；
#   3. 保存后执行 nginx -t && nginx -s reload。
#
# 说明：
#   - 本段不含 listen / server_name / ssl_*，直接粘进面板站点配置的
#     server 块里即可，端口与证书保持面板原样（两个域名都走
#     Cloudflare 时，源站甚至可以只用 80，TLS 在边缘终结）。
#   - 若面板给每个站点各生成一份配置，把「站点 A 段」「站点 B 段」
#     分别拆开粘过去也能跑；本文件按域名自动分流，不拆同样成立。
#   - 域名以后若再变，只改文件顶部的两处域名即可。
=====================================================================

# ---------- 目录分流：命中 StyleCool 域名就用 stylecool 目录 ----------
set $approot "functioncool";
if ($host = 'stylecool.mutantcat.org')     { set $approot "stylecool"; }
if ($host = 'www.stylecool.mutantcat.org') { set $approot "stylecool"; }
if ($host = 'style.functioncool.xyz')      { set $approot "stylecool"; }

root /www/wwwroot/$approot;
index index.php;

# ---------- 旧域名统一 301 到新域 ----------
if ($host = 'functioncool.xyz')               { return 301 https://functioncool.mutantcat.org$request_uri; }
if ($host = 'www.functioncool.xyz')           { return 301 https://functioncool.mutantcat.org$request_uri; }
if ($host = 'www.functioncool.mutantcat.org') { return 301 https://functioncool.mutantcat.org$request_uri; }
if ($host = 'style.functioncool.xyz')         { return 301 https://stylecool.mutantcat.org$request_uri; }
if ($host = 'www.stylecool.mutantcat.org')    { return 301 https://stylecool.mutantcat.org$request_uri; }

# ---------- StyleCool 独立成站，主站上的旧路径整体搬过去 ----------
if ($approot = 'functioncool') {
    rewrite ^/stylecool(/.*)?$ https://stylecool.mutantcat.org$1 permanent;
    rewrite ^/example/?$       https://stylecool.mutantcat.org/example permanent;
}

location / {
    try_files $uri @rewriteapp;
}

location @rewriteapp {
    # 站点 A：FunctionCool 专属
    if ($approot = 'functioncool') {
        rewrite ^/sitemap\.xml$ /sitemap.php last;
        rewrite ^/404/?$        /404.php last;
    }

    # 站点 B：StyleCool 专属
    if ($approot = 'stylecool') {
        # 旧写法把 StyleCool 挂在主站 /stylecool/ 下，这里把多余前缀去掉
        rewrite ^/stylecool/?(.*)$ /$1 permanent;
        rewrite ^/skillapi/?$ /index.php last;
        rewrite ^/example/?$  /example.php last;
    }

    # 两个站点共用的兜底：一律交给 index.php
    # （脚本用 REQUEST_URI 取原始路径自行路由，不需要 PATH_INFO，
    #   也不会出现 /index.php/... 反复重写的循环）
    rewrite ^(.*)$ /index.php last;
}

location ~ \.php$ {
    try_files $uri =404;
    fastcgi_pass unix:/var/run/php/php-fpm.sock;
    fastcgi_index index.php;
    include fastcgi_params;
    fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;
}

error_page 404 /404.php;

---------------------------------------------------------------------
# 说明
# - /sitemap.xml 只有 FunctionCool 有；StyleCool 站点没有 sitemap.php，
#   该项只在站点 A 生效。
# - /search、/mcpapi、/notes/NNN 由 FunctionCool 的 index.php 自行处理
#   （含 /mcpapi -> /skillapi 的 301），nginx 侧不写规则。
# - /example、/skillapi 只在 StyleCool 有；站点 A 上的这两个旧路径已在
#   上面 301 到 stylecool.mutantcat.org。
# - StyleCool 目录里没有 404.php，该站点的 404 会落到 nginx 默认页；
#   想让 StyleCool 的 404 也显示文档页，把这行改成 error_page 404 /index.php;
#   即可（FunctionCool 会因此丢掉自己的 404.php，二选一）。
